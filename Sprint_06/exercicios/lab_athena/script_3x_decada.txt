
-- os 3 nomes mais usados em cada decada

WITH Decadas AS (
    SELECT
        nome,
        FLOOR(ano / 10) * 10 AS decada, -- intervalos 10anos
        SUM(total) AS total_decada -- soma por decada
    FROM meubanco.tabelanomes
    WHERE ano >= 1950 -- apartir 1950
    GROUP BY nome, FLOOR(ano / 10) * 10
),
RankedNomes AS (
    SELECT
        nome,
        decada,
        total_decada,
        ROW_NUMBER() OVER (PARTITION BY decada ORDER BY total_decada DESC) AS posicao
    FROM Decadas
)
SELECT
    decada,
    nome,
    total_decada
FROM RankedNomes
WHERE posicao <= 3 -- apresenta 3mais por decada
ORDER BY decada, posicao;
